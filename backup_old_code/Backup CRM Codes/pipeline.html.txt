function renderTimeline(opportunities) {
  const today = new Date(new Date().toLocaleString("en-US", { timeZone: "Asia/Bangkok" }));
  const timelineContainer = document.getElementById('timeline-container');
  timelineContainer.innerHTML = ''; // Clear existing bullets

  const futureOpportunities = opportunities
    .map(item => {
      lockStateMap[item['Unique-ID']] = item['Loading-Check']; // Initialize lock state map

      const goodsLoadingDate = new Date(item['Goods-Loading-Date'].replace(/(\d+)\/(\d+)\/(\d+)/, '$2/$1/$3'));
      const remainingDays = Math.ceil((goodsLoadingDate - today) / (1000 * 60 * 60 * 24));

      const options = { weekday: 'short', day: 'numeric', month: 'short', year: '2-digit' };
      const formattedDate = goodsLoadingDate.toLocaleDateString('en-GB', options).replace(',', '');

      return { ...item, goodsLoadingDate, remainingDays, formattedDate };
    })
    .filter(item => item.goodsLoadingDate > today) // Keep only future dates
    .sort((a, b) => a.goodsLoadingDate - b.goodsLoadingDate); // Sort by date

  let currentMonth = ''; // Track the current month to insert month-lines
  futureOpportunities.forEach(item => {
    const goodsLoadingMonth = item.goodsLoadingDate.toLocaleDateString('en-GB', { month: 'short', year: 'numeric' });
    
    // Insert a month-line when the month changes
    if (goodsLoadingMonth !== currentMonth) {
      currentMonth = goodsLoadingMonth;
      
      // Create and append the month-line
      const monthLine = document.createElement('div');
      monthLine.classList.add('month-line');
      monthLine.textContent = currentMonth;
      timelineContainer.appendChild(monthLine);
    }

    // Create and append the bullet
    const bullet = document.createElement('div');
    bullet.classList.add('bullet');
    bullet.setAttribute('data-unique-id', item['Unique-ID']);

    const isChecked = lockStateMap[item['Unique-ID']] === 'Checked';
    const tagText = item['Tags'] || ''; // Get tag text or set empty if none
    bullet.innerHTML = `
      <div class="project-info">
        <span class="bullet-project-name">${item['Project']}</span><br>
        ${item.formattedDate}<br>
        ${item['Topic']}
      </div>
      <div class="remaining-days">
        ${item.remainingDays}
      </div>
      <i class="fa-solid fa-lock check-icon" style="display: ${isChecked ? 'block' : 'none'};"></i>
      ${tagText ? `<input type="text" class="bullet-tag-input" value="${tagText}" readonly />` : ''}
    `;

    bullet.addEventListener('click', () => toggleLoadingCheck(bullet));
    timelineContainer.appendChild(bullet);
  });
}

document.querySelector('.hotcommand').addEventListener('input', function(e) {
  var value = this.value;
  var suggestionBox = document.getElementById('hotcommand-suggestions');
  
  // Hide suggestions when a space is pressed (and no dropdown item is selected)
  if (value.indexOf(' ') !== -1 && value.charAt(0) === '@') {
    suggestionBox.style.display = 'none';
    return;
  }
  
  if (value.charAt(0) === '@') {
    var partialName = value.substring(1).split(' ')[0];
    var owners = getCardOwnerList(); // Returns unique names
    var filtered = owners.filter(function(name) {
      return name.toLowerCase().startsWith(partialName.toLowerCase());
    });
    
    suggestionBox.innerHTML = '';
    if (filtered.length > 0) {
      filtered.forEach(function(name) {
        var suggestionItem = document.createElement('div');
        suggestionItem.className = 'suggestion-item';
        suggestionItem.textContent = name;
        suggestionItem.addEventListener('click', function() {
          var remainder = value.indexOf(' ') !== -1 ? value.substring(value.indexOf(' ')) : '';
          document.querySelector('.hotcommand').value = '@' + name + remainder;
          suggestionBox.style.display = 'none';
        });
        suggestionBox.appendChild(suggestionItem);
      });
      suggestionBox.style.display = 'block';
    } else {
      suggestionBox.style.display = 'none';
    }
  } else {
    suggestionBox.style.display = 'none';
  }
});

function toggleLoadingCheck(bullet) {
  if (busy) return;

  busy = true; // Set busy flag to prevent multiple clicks
  const uniqueId = bullet.getAttribute('data-unique-id');
  const icon = bullet.querySelector('.check-icon');

  const isCurrentlyChecked = icon.style.display === 'block';
  const newStatus = isCurrentlyChecked ? '' : 'Checked';
  const message = newStatus === 'Checked' ? "Locked" : "Removed Lock";

  // Display temporary message to indicate locking/unlocking
  displayTemporaryMessage(message);

  // Update the icon immediately for visual feedback
  icon.style.display = newStatus ? 'block' : 'none';
  lockStateMap[uniqueId] = newStatus;

  // Perform backend update
  google.script.run.withSuccessHandler(() => {
    busy = false; // Reset busy flag once update is confirmed

    // Update lock icons in both calendar and timeline
    updateAllCalendarIcons(uniqueId, newStatus);
    updateTimelineIcon(uniqueId, newStatus);
  }).updateLoadingCheck(uniqueId, newStatus);
}


function updateAllCalendarIcons(uniqueId, status) {
  const calendarElements = document.querySelectorAll(`[data-id="${uniqueId}"]`);
  calendarElements.forEach(calendarElement => {
    const icon = calendarElement.querySelector('.fa-lock');
    if (icon) {
      icon.style.display = status === 'Checked' ? 'inline-block' : 'none';
      calendarElement.setAttribute('draggable', status === 'Checked' ? 'false' : 'true');
    }
  });
}

// -----------------------------  Function to Generate Each Card -----------------------------
function generateCards(opportunities) {
  const statuses = ['Doing', 'Qualified', 'Quoted', 'Following', 'End'];
  const statusCounts = { 'Doing': 0, 'Qualified': 0, 'Quoted': 0, 'Following': 0, 'End': 0 };
  const statusAmounts = { 'Doing': 0, 'Qualified': 0, 'Quoted': 0, 'Following': 0, 'End': 0 };
  const endColumnCards = []; // Store cards for the "End" column for sorting
  const columnHtml = { 'Doing': '', 'Qualified': '', 'Quoted': '', 'Following': '', 'End': '' };

  const contactsDataMap = {};
  contactsData.forEach(contact => {
    if (contact['Perma-Link']) contactsDataMap[contact['Project']] = contact['Perma-Link'];
    if (contact['Email']) contactsDataMap[contact['Project'] + '_Email'] = contact['Email'];
  });

  opportunities.forEach(opportunity => {
    const endStatuses = ['Completed', 'Discarded', 'Won', 'Qualified-Loss', 'Quoted-Loss', 'Following-Loss'];
    let status = opportunity['Status'];
    const isEndColumn = endStatuses.includes(status);

    if (isEndColumn) {
      statusCounts['End']++;
      status = 'End';
    } else if (!statuses.includes(status)) {
      return; 
    } else {
      statusCounts[status]++;
    }

    const amount = parseFloat(opportunity['Amount']) || 0;
    statusAmounts[status] += amount;
    const formattedAmount = formatAmountOneDecimal(amount);
    const coinIconHtml = formattedAmount ? `<i class="fa-duotone fa-solid fa-coins coin-icon"></i>` : '';



    // Also compute the full and short versions for hover effect:
    const fullAmount = amount > 0 ? amount.toLocaleString('en-US') : '';
    const shortAmount = formatAmountOneDecimal(amount);



    const calendarCheckIconHtml = getRoundCheckIcon(opportunity['Follow-Up-History']);
    const dueDateIconHtml = getDueDateIcon(opportunity['Due-Date'], isEndColumn);
    const followUpHistoryFormatted = formatFollowUpHistory(opportunity['Follow-Up-History']);
    const docfileIconHtml = opportunity['Doc-Unique-ID'] ? `<i class="fa-solid fa-square-poll-horizontal"></i>` : '';
    const textCardHtml = `
      <div class="text-card">
        <textarea …></textarea>
      </div>
      <div class="card-search-wrapper" style="display:none;">
        <input type="text" class="card-search"  placeholder="SEARCH" autocomplete="off"/>
        <div class="card-search-count">(0)</div>
      </div>
    `;

    const cardStyle = status === 'End' ? 'style="background-color: #DBF2F2;"' : '';
    const linkIconHtml = opportunity['Link'] ? `<a href="${opportunity['Link']}" target="_blank"><i class="fa-solid fa-square-arrow-up-right"></i></a>` : '';
    const cardOwnerHtml = opportunity['Card-Owner'] ? `<div class="card-owner-style">${calendarCheckIconHtml}${opportunity['Card-Owner']}</div>` : '';
    const goodsLoadingDateIconHtml = getGoodsLoadingDateIcon(opportunity['Goods-Loading-Date'], isEndColumn);
    const permaLink = contactsDataMap[opportunity['Project']] || '';
    const permaLinkIconHtml = permaLink ? `<a href="${permaLink}" target="_blank"><i class="fa-brands fa-google-drive"></i></a>` : '';
    const email = contactsDataMap[opportunity['Project'] + '_Email'] || '';
    const gmailIconHtml = email ? `<a href="javascript:void(0);" onclick="searchInGmail('${email}')"><i class="fa-solid fa-envelope"></i></a>` : '';


    let oemHtml = "";
    if (opportunity['Topic'].includes("#OEM")) {
      const progress = getOemProgress(opportunity['Unique-ID']);
      if (progress && progress.indexOf("%") !== -1) {
        const pct = parseInt(progress.replace("%", ""), 10);
        let iconColorClass = (pct === 100) ? "oem-complete" : "oem-incomplete fa-bounce";
        oemHtml = `<span class="oem-progress-text" onclick="showOemModal('${opportunity['Unique-ID']}')">${progress}</span>
                  <i class="fa-solid fa-flask oem-icon ${iconColorClass}" onclick="showOemModal('${opportunity['Unique-ID']}')"></i>`;
      } else {
        oemHtml = `<i class="fa-solid fa-plus oem-icon" onclick="showOemModal('${opportunity['Unique-ID']}')"></i>`;
      }
    }


    const cardHtml = `
      <div id="card-${opportunity['Unique-ID']}" class="card" ${cardStyle} 
          data-type="${opportunity['Type']}"
          data-status="${opportunity['Status']}"
          data-project="${opportunity['Project']}"
          data-card-owner="${opportunity['Card-Owner']}"
          data-created-date="${opportunity['Created-Date']}"
          data-reserve="${opportunity['Reserve']}"
          data-invoice="${opportunity['Invoice']}"
          data-goods-ready-date="${opportunity['Goods-Ready-Date']}"
          data-goods-loading-date="${opportunity['Goods-Loading-Date']}"
          data-quoted-date="${opportunity['Quoted-Date']}"
          data-amount="${opportunity['Amount']}"
          data-topic="${opportunity['Topic']}"
          data-due-date="${opportunity['Due-Date']}"
          data-follow-up-history="${opportunity['Follow-Up-History']}"
          data-comment="${opportunity['Comment']}"
          data-link="${opportunity['Link']}"
          data-loss-reason="${opportunity['Loss-Reason']}"
          data-unique-id="${opportunity['Unique-ID']}"
          data-doc-unique-id="${opportunity['Doc-Unique-ID']}"
          data-gantt-status="${opportunity['Gantt-Status']}"
          data-fda-started-date="${opportunity['FDA-Started-Date']}"
          data-special-card="${opportunity['Special-Card']}"
          data-tag="${opportunity['Tags']}">

          <div class="card-tag">
            <input type="text" class="tag-input ${opportunity['Tags'] ? 'has-text' : ''}" 
                  value="${opportunity['Tags']}" 
                  onkeypress="handleTagChange(event, '${opportunity['Unique-ID']}')" />
          </div>

          <div class="oem-progress-box">
            ${oemHtml}
          </div>

          <div class="amount-box">
            ${coinIconHtml}
            <div class="amount-values">
              <div class="formatted-amount" data-full="${fullAmount}" data-short="${shortAmount}">
                <span class="short-form">${shortAmount}</span>
                <span class="full-form">${fullAmount}</span>
              </div>
            </div>
          </div>


        <div class="upper-card">
          <div class="low-uppercard">
            <div class="project-name" onclick="copyToClipboard(this, event)">${opportunity['Project']}</div>
            <div class="topic-name" onclick="copyToClipboard(this, event)">${opportunity['Topic']}</div>
          </div>
        </div>
        ${textCardHtml}
        <div class="middle-card">
          ${followUpHistoryFormatted}
        </div>
        <div class="lower-card">
          <div class="low-lowercard">
            <div class="icon-group-left">${cardOwnerHtml}</div>
            <div class="icon-group-right">${dueDateIconHtml}${goodsLoadingDateIconHtml}${permaLinkIconHtml}${linkIconHtml}${gmailIconHtml}${docfileIconHtml}</div>
          </div>
        </div>
      </div>`;


    if (status !== 'End') {
      columnHtml[status] += cardHtml;
    } else {
      endColumnCards.push({ html: cardHtml, newestDate: extractNewestDate(opportunity['Follow-Up-History']) });
    }
  });

  statuses.forEach(status => {
    const column = document.getElementById(status);
    if (column && columnHtml[status]) {
      column.innerHTML = `<h3>${status} (${statusCounts[status]})</h3><div class="amount"></div>${columnHtml[status]}`;
    }
  });

  // Populate the "End" column
  const endColumn = document.getElementById('End');
  if (endColumn) {
    endColumnCards.sort((a, b) => b.newestDate - a.newestDate);
    endColumn.innerHTML = `<h3>End (${statusCounts['End']})</h3><div class="amount"></div>` + endColumnCards.map(card => card.html).join('');
  }

  updateStatusHeaders(statusCounts, statusAmounts);
  updateCardColors();
  infoSidebarEventListeners();
  expandMiddleClass();
  autoTextarea();
  setupCardSearch();
}


// Helper function to format amounts
function formatAmountOneDecimal(amount) {
  amount = parseFloat(amount);
  if (isNaN(amount) || amount === 0) return ''; // Return blank if amount is NaN or 0
  
  let formatted;
  if (amount >= 1000000) {
    formatted = (amount / 1000000).toFixed(1);
    if (formatted.endsWith('.00')) {
      formatted = (amount / 1000000).toFixed(0);
    }
    return formatted + 'M';
  } else if (amount >= 1000) {
    formatted = (amount / 1000).toFixed(1);
    if (formatted.endsWith('.00')) {
      formatted = (amount / 1000).toFixed(0);
    }
    return formatted + 'K';
  } else {
    formatted = amount.toFixed(1);
    if (formatted.endsWith('.00')) {
      formatted = amount.toFixed(0);
    }
    return formatted;
  }
}

// ----------------------------- Extract the Newest Date from follow-up history for Cards in End Column -------------------------
function extractNewestDate(followUpHistory) {
  let newestDate = null;
  const dateRegex = /\[(\w{3}\s*\d{1,2},?\s*\d{4})\]/g;
  let match;

  while ((match = dateRegex.exec(followUpHistory)) !== null) {
    const dateStr = match[1];
    const formattedDateStr = dateStr.replace(/(\w{3})(\d{1,2}),(\d{4})/, "$1 $2,$3");
    const date = new Date(formattedDateStr);

    if (!isNaN(date) && (!newestDate || date > newestDate)) {
      newestDate = date;
    }
  }
  return newestDate || new Date(0); // Return a default old date if none found
}

// -----------------------------  Resize Textarea and Update Textarea to Google Sheet -----------------------------
function autoTextarea() {
  document.querySelectorAll('.text-card textarea').forEach(function(textarea) {
    resizeTextarea(textarea); // Resize on load
    textarea.addEventListener('input', function() {
      resizeTextarea(this); // Resize on input
    });
    textarea.addEventListener('keydown', function(event) {
      if (event.key === 'Enter' && event.shiftKey) {
        // Shift+Enter was pressed, insert a newline at cursor position
        event.preventDefault();
        insertAtCursor(textarea, '\n');
        resizeTextarea(textarea);
      } else if (event.key === 'Enter') {
        // Enter was pressed without Shift, handle submission
        event.preventDefault();
        const textValue = textarea.value; // Capture textarea value
        textarea.value = ''; // Clear the textarea
        resizeTextarea(textarea);
        UpdateText(this, textValue);
      }
    });
  });
}

function insertAtCursor(textarea, text) {
  const startPos = textarea.selectionStart;
  const endPos = textarea.selectionEnd;
  textarea.value = textarea.value.substring(0, startPos)
    + text
    + textarea.value.substring(endPos, textarea.value.length);
  textarea.selectionStart = startPos + text.length;
  textarea.selectionEnd = startPos + text.length;
}

// -----------------------------  Function to format the follow-up history text -----------------------------
function formatFollowUpHistory(text) {
  // Replace hashtags with formatted tags
  text = text.replace(/#(\w+)/g, "<span class='hashtag'>#$1</span>");

  // Split the text into lines
  const lines = text.trim().split('\n');

  // Add a line break above each line that starts with a date format
  const formattedLines = lines.map((line, index) => {
    if (/^\[\w{3} \d{2}, \d{4}\]/.test(line)) {
      return (index > 0 ? '<br>' : '') + line.replace(/(\[\w{3} \d{2}, \d{4}\])(.*)/, function(match, datePart, rest) {
        return `<span class='date-part'>${datePart}</span>${rest}`;
      });
    }
    return line;
  });

  // Join the formatted lines with a single line break
  return formattedLines.join('<br>');
}

// ----------------------------- Red Card Color Based On follow-up History Date and Arrange to Top Area -------------
function updateCardColors() {
  var today = new Date();
  today.setHours(0, 0, 0, 0); // Reset time

  document.querySelectorAll('.card').forEach(function(card) {
    // --- Special card handling ---
    if (card.getAttribute('data-special-card') === 'Special') {
      var followUpHistory = card.getAttribute('data-follow-up-history');
      let newestDate = null;
      let dateCount = 0;
      if (followUpHistory) {
        const dateRegex = /\[(\w{3}\s*\d{1,2},?\s*\d{4})\]/g;
        let match;
        while ((match = dateRegex.exec(followUpHistory)) !== null) {
          dateCount++;
          const dateStr = match[1];
          const formattedDateStr = dateStr.replace(/(\w{3})(\d{1,2}),(\d{4})/, "$1 $2,$3");
          const date = new Date(formattedDateStr);
          if (!isNaN(date) && (!newestDate || date > newestDate)) {
            newestDate = date;
          }
        }
      }
      // Apply the special-card class if:
      // 1. The newest date is before today, OR
      // 2. The newest date is today or in the future AND there is only one date in the follow-up history.
      if (newestDate && (newestDate.getTime() < today.getTime() || (newestDate.getTime() >= today.getTime() && dateCount === 1))) {
        card.classList.add('special-card');
      } else {
        card.classList.remove('special-card');
      }
      // Skip the normal red/yellow logic for special cards.
      return;
    }

    // --- Existing red/yellow logic for non-special cards ---
    var status = card.getAttribute('data-status');
    var endStatuses = ['Completed', 'Discarded', 'Won', 'Qualified-Loss', 'Quoted-Loss', 'Following-Loss'];
    if (endStatuses.includes(status)) {
      return;
    }

    var followUpHistory = card.getAttribute('data-follow-up-history');
    var dueDateStr = card.getAttribute('data-due-date');
    var dueDate = dueDateStr ? parseDueDate(dueDateStr) : null;
    var isDueOrPast = checkDueDateOrPast(card);

    let newestDate = null;
    if (followUpHistory) {
      const dateRegex = /\[(\w{3}\s*\d{1,2},?\s*\d{4})\]/g;
      let match;
      while ((match = dateRegex.exec(followUpHistory)) !== null) {
        const dateStr = match[1];
        const formattedDateStr = dateStr.replace(/(\w{3})(\d{1,2}),(\d{4})/, "$1 $2,$3");
        const date = new Date(formattedDateStr);
        if (!isNaN(date) && (!newestDate || date > newestDate)) {
          newestDate = date;
        }
      }
    }

    if (newestDate) {
      const diffTime = Math.abs(today - newestDate);
      const diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24));

      // Apply red/yellow classes based on the diffDays
      if ((diffDays > 2 && (!dueDate || dueDate <= today)) || isDueOrPast) {
        card.classList.add('red-card');
      } else if (diffDays === 2 && (!dueDate || dueDate <= today)) {
        card.classList.add('yellow-card');
      } else {
        card.classList.remove('red-card', 'yellow-card');
      }
    }
  });

  // --- Sorting logic ---
  var columns = document.querySelectorAll('.status-columns .column');

  // Helper function to assign a "rank" to each card:
  // 0 = special-card, 1 = red-card, 2 = yellow-card, 3 = normal card
  function getCardRank(card) {
    if (card.classList.contains('special-card')) return 0;
    if (card.classList.contains('red-card'))     return 1;
    if (card.classList.contains('yellow-card'))  return 2;
    return 3;
  }

  columns.forEach(column => {
    var sortedCards = Array.from(column.getElementsByClassName('card'));
    
    // Sort by rank: special > red > yellow > normal
    sortedCards.sort((a, b) => {
      return getCardRank(a) - getCardRank(b);
    });
    
    sortedCards.forEach(card => column.appendChild(card));
  });
}



// ------------------- Check if the Due Date Icon is Present and Indicating Due or Past -------------------
function checkDueDateOrPast(card) {
  var bellIcon = card.querySelector('.fa-bell.fa-bounce');
  return bellIcon !== null;
}

//Helper function to parse due dates in the format "dd/mm/yyyy"
function parseDueDate(dueDateStr) {
  var parts = dueDateStr.split('/');
  // Note: Months are 0-indexed in JavaScript Date
  return new Date(parts[2], parts[1] - 1, parts[0]);
}

// ------------------- Display Bell Icon if It has Due Date Due Date----------------
function getDueDateIcon(dueDateStr, isEndColumn) {
  if (isEndColumn) {
    return '';
  }

  if (dueDateStr && dueDateStr.trim() !== '') {
    var today = new Date();
    today.setHours(0, 0, 0, 0); // Set today's date to midnight for comparison

    var dueDate = parseDueDate(dueDateStr);
    var formattedDueDate = formatDate(dueDate);

    // Determine if the due date is today or has passed
    var isDueOrPast = dueDate <= today;
    var bellIconHtml = isDueOrPast 
      ? '<i class="fa-solid fa-bell fa-bounce" style="color: #e76e79; font-size: 20px;"></i>'
      : '<i class="fa-solid fa-bell"></i>';

    return `<span class="due-date-container">${bellIconHtml} <span class="due-date-text">${formattedDueDate}</span></span>`;
  }
  return '';
}


function formatDate(date) {
  var day = date.getDate().toString().padStart(2, '0');
  var month = (date.getMonth() + 1).toString().padStart(2, '0'); // Months are 0-indexed
  var year = date.getFullYear();
  return day + '/' + month; 
}

// ----------------- Display Check icon if the newest date in follow-up history is today----------------
function getRoundCheckIcon(followUpHistory) {
  var today = new Date();
  today.setHours(0, 0, 0, 0); // Set today's date to midnight for accurate comparison

  if (isNewestDateToday(followUpHistory, today)) {
    return '<i class="fa-solid fa-circle-check"></i>'; 
  }
  return '';
}

// Helper function to check if the newest date in follow-up history is today
function isNewestDateToday(followUpHistory, today) {
  const dateRegex = /\[(\w{3}\s*\d{1,2},?\s*\d{4})\]/g;
  let match, newestDate = null;

  while ((match = dateRegex.exec(followUpHistory)) !== null) {
    const dateStr = match[1];
    const date = new Date(dateStr);
    if (!newestDate || date > newestDate) {
      newestDate = date;
    }
  }

  return newestDate && (newestDate.setHours(0, 0, 0, 0) === today.getTime());
}

// ------------------------------------Setup Search Card------------------------------------------
function navigateHighlights(card, direction = 1) {
  const highlights = Array.from(card.querySelectorAll('mark.highlight-search'));
  if (!highlights.length) return;

  // clear old “current-highlight”
  highlights.forEach(m => m.classList.remove('current-highlight'));

  // initialize or advance index
  if (typeof card._searchIndex !== 'number') {
    // start before first if going forward, or after last if going backward
    card._searchIndex = direction > 0 ? -1 : highlights.length;
  }
  let newIndex = card._searchIndex + direction;
  // wrap
  if (newIndex < 0)            newIndex = highlights.length - 1;
  else if (newIndex >= highlights.length) newIndex = 0;

  card._searchIndex = newIndex;

  // scroll into view within column
  const el = highlights[newIndex];
  const column = card.closest('.column');
  if (column) {
    const top = el.offsetTop - column.clientHeight / 2;
    column.scrollTo({ top, behavior: 'smooth' });
  } else {
    el.scrollIntoView({ behavior: 'smooth', block: 'center' });
  }

  // mark it “current-highlight”
  el.classList.add('current-highlight');
}


function setupCardSearch() {
  document.querySelectorAll('.card').forEach(card => {
    const input     = card.querySelector('.card-search');
    const countSpan = card.querySelector('.card-search-count');
    if (!input || !countSpan) return;

    // ensure hidden on load
    countSpan.style.color = 'transparent';

    input.addEventListener('input', () => {
      card._searchIndex = -1;
      const term = input.value.trim();
      const marker = new Mark(card);

      marker.unmark({
        done: () => {
          let count = 0;
          if (term) {
            marker.mark(term, {
              separateWordSearch: false,
              className: 'highlight-search',
              done: matched => { count = matched; }
            });
          }
          if (!count) {
            count = card.querySelectorAll('mark.highlight-search').length;
          }
          countSpan.textContent = `(${count})`;

          // flip color based on whether there's text
          if (term) {
            countSpan.style.color = 'rgba(0,0,0,0.5)';
          } else {
            countSpan.style.color = 'transparent';
          }
        }
      });
    });

    input.addEventListener('keydown', e => {
      if (e.key === 'Enter' || e.key === 'ArrowDown') {
        e.preventDefault();
        navigateHighlights(card,  1);
      } else if (e.key === 'ArrowUp') {
        e.preventDefault();
        navigateHighlights(card, -1);
      }
    });
  });
}

      }).getAllAuthorizedUsers();

      var contactData = findContactData(card.getAttribute('data-project'));
      showInfoSidebarWithData(cardData, contactData);

      var infoSidebar = document.getElementById('info-sidebar');
      infoSidebar.setAttribute('data-unique-id', cardData.uniqueId);

      // Open the info sidebar
      infoSidebar.style.width = '300px';
      // --- accordion collapse/expand for .left-section-box ---
      const sections = infoSidebar.querySelectorAll('.left-section-box');
      sections.forEach(box => {
        const header = box.querySelector('h3');
        // hide all content except the header
        Array.from(box.children).slice(1).forEach(child => {
          child.style.display = 'none';
        });
        header.style.cursor = 'pointer';
        header.addEventListener('click', () => {
          // collapse everyone else
          sections.forEach(otherBox => {
            if (otherBox !== box) {
              Array.from(otherBox.children).slice(1).forEach(child => {
                child.style.display = 'none';
              });
            }
          });

          // now toggle this one
          Array.from(box.children).slice(1).forEach(child => {
            if (child.style.display === 'none') {
              // remove inline display—allow stylesheet (e.g. display:flex) to apply
              child.style.removeProperty('display');
            } else {
              child.style.display = 'none';
            }
          });
        });
      });
      document.body.classList.add('info-sidebar-open'); // Add a class to indicate the sidebar is open
    });
  });
}

//----------- Function to get button options based on the current status--------------
function getButtonOptionsBasedOnStatus(status) {
    switch (status) {
        case 'Doing':
            return ['Completed', 'Discarded'];
        case 'Qualified':
            return ['Quoted', 'Qualified-Loss'];
        case 'Quoted':
            return ['Following', 'Quoted-Loss'];
        case 'Following':
            return ['Won', 'Following-Loss'];
        default:
            return []; //
    }
}

function updateStatusAndCloseSidebar(uniqueId, newStatus) {
  var lossReasonInput = '';
  var inputAmount = '';
  var cardElement = document.querySelector(`.card[data-unique-id="${uniqueId}"]`);

  // For statuses that require dates (example: Won)
  if (newStatus === 'Won') {
    var goodsLoadingDate = goodsLoadingDatePicker.selectedDates[0];
    var goodsReadyDate = goodsReadyDatePicker.selectedDates[0];
    if (!goodsLoadingDate || !goodsReadyDate) {
      alert('Please enter both "Goods Loading Date" and "Goods Ready Date" before marking as Won.');
      return;
    }
  }

  // For loss statuses, require a loss reason input.
  if (['Discarded', 'Qualified-Loss', 'Quoted-Loss', 'Following-Loss'].includes(newStatus)) {
    lossReasonInput = document.getElementById('lossReason').value.trim();
    if (lossReasonInput === '') {
      alert('Please enter the "Loss Reason".');
      return;
    }
  }

  // For the "Quoted" status, require a valid amount.
  if (newStatus === 'Quoted') {
    inputAmount = document.getElementById('inputAmount').value.trim();
    if (inputAmount === '' || isNaN(inputAmount)) {
      alert('Please enter a valid amount.');
      return;
    }
  }

  // Get displayName from the server and update the card.
  google.script.run.withSuccessHandler(function(displayName) {
    if (cardElement) {
      // Update card status attribute and owner.
      cardElement.setAttribute('data-status', newStatus);
      var targetColumn = (newStatus === 'Won' || newStatus === 'Completed' ||
                          newStatus === 'Discarded' || newStatus.indexOf('-Loss') > -1) ? 'End' : newStatus;
      var columnElement = document.getElementById(targetColumn);
      if (columnElement && cardElement.parentNode) {
        cardElement.parentNode.removeChild(cardElement);
        columnElement.appendChild(cardElement);
      }
      cardElement.querySelector('.icon-group-left').innerHTML = `<div class="card-owner-style">${displayName}</div>`;

      // Generate the formatted follow-up history string on the client side.
      var currentDate = new Date();
      var day = String(currentDate.getDate()).padStart(2, '0');
      var monthNames = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
      var year = currentDate.getFullYear();
      var rawHistory = "";
      if (["Completed", "Quoted", "Following", "Won"].includes(newStatus)) {
        rawHistory = "[" + monthNames[currentDate.getMonth()] + " " + day + ", " + year + "] [" + newStatus + " by '" + displayName + "']";
      } else if (["Discarded", "Qualified-Loss", "Quoted-Loss", "Following-Loss"].includes(newStatus)) {
        rawHistory = "[" + monthNames[currentDate.getMonth()] + " " + day + ", " + year + "] [" + newStatus + " by '" + displayName + "'] [" + lossReasonInput + "]";
      }
      
      // Bold the date portion using a regex replacement.
      var formattedFollowUpHistory = rawHistory.replace(
        /\[(\w{3}\s\d{1,2},\s\d{4})\]/,
        '<span style="font-weight: 600;">[$1]</span>'
      );
      
      // Update the middle-card element without removing existing history.
      var middleCardEl = cardElement.querySelector('.middle-card');
      if (middleCardEl) {
        var existingHistory = middleCardEl.innerHTML.trim();
        var newHistory = formattedFollowUpHistory + '<br><br>' + existingHistory;
        middleCardEl.innerHTML = newHistory;
        cardElement.setAttribute('data-follow-up-history', newHistory);
      }
      
      // --- Update the Amount Box UI if the new status is "Quoted" ---
      if (newStatus === 'Quoted' && inputAmount) {
        var amountElement = cardElement.querySelector('.formatted-amount');
        if (amountElement) {
          var newAmount = parseFloat(inputAmount);
          // Use your helper to get the short formatted value.
          var formattedShort = formatAmountOneDecimal(newAmount);
          var formattedFull = newAmount.toLocaleString('en-US');
          // Update innerHTML to include the coin icon and the spans.
          amountElement.innerHTML = `<i class="fa-duotone fa-solid fa-coins coin-icon"></i><span class="short-form"> ${formattedShort}</span><span class="full-form"> ${formattedFull}</span>`;
          amountElement.setAttribute('data-full', formattedFull);
          amountElement.setAttribute('data-short', formattedShort);
        }
      }
      
      // Call updateCardColors() so that the card CSS is re-evaluated
      // based on the new follow-up date (which is the current date) and amount updates.
      updateCardColors();
    }

    // Persist the changes on the server side.
    google.script.run.updateOpportunityStatusAndLossReasonAndAmount(uniqueId, newStatus, lossReasonInput, inputAmount, displayName);
  }).authorizedUser();

  displayTemporaryMessage("Status Modified Successfully.");
  closeInfoSidebar();
}

//----------- CMD + E to toggle the #OEM --------------
document.addEventListener('keydown', function (event) {
    // Check for CMD+E on macOS or CONTROL+E on Windows
    if ((event.metaKey || event.ctrlKey) && event.key.toLowerCase() === 'e') {
        event.preventDefault(); // Prevent default behavior (optional, depends on your app's needs)
        toggleOEMFilter();
    }
});

let oemFilterEnabled = false; // Track filter toggle state

function toggleOEMFilter() {
    oemFilterEnabled = !oemFilterEnabled; // Toggle the filter state

    document.querySelectorAll('.card').forEach(function (card) {
        var cardTopic = card.getAttribute('data-topic') || ''; // Get the card's topic name

        if (oemFilterEnabled) {
            if (cardTopic.includes('#OEM')) {
                card.style.display = 'block'; // Show cards with "#OEM"
            } else {
                card.style.display = 'none'; // Hide other cards
            }
        } else {
            card.style.display = 'block'; // Show all cards when filter is disabled
        }
    });

    // Count only visible cards that are NOT in the "End" column.
    const visibleCards = Array.from(document.querySelectorAll('.card')).filter(card => {
        // card.style.display !== 'none' means it's visible
        // Exclude cards that are inside an element with id "End"
        return card.style.display !== 'none' && !card.closest('#End');
    }).length;

    const countCardElement = document.getElementById('countCard');
    if (countCardElement) {
        countCardElement.innerText = visibleCards;
    }
}

  if (key === 's' || key === 'ห') {
    // Check if the event target is NOT an input or textarea
    const tagName = event.target.tagName;
    if (tagName !== 'INPUT' && tagName !== 'TEXTAREA') {
      const searchbar = document.querySelector('.searchbarpipeline');
      if (searchbar) {
        searchbar.focus();
        event.preventDefault();
      }
    }
  }
});

      if (cardText.includes(searchQuery) && (ownerName === '' || cardOwner === ownerName)) {
        card.style.display = '';
      } else {
        card.style.display = 'none';
      }
      
      // Highlight the search query in the card using mark.js
      var instance = new Mark(card);
      instance.unmark({
        done: function() {
          if (searchQuery !== '') {
            instance.mark(searchQuery, {
              separateWordSearch: false,
              className: "highlight-search"
            });
          }
        }
      });
    });
    updateCardCount(); // Update the card count after filtering
  }, 300);
}

function cardSubmit() {

  if (projectName && cardType && topic) {
    // Generate a temporary unique ID for immediate UI feedback.
    var tempUniqueID = 'temp-' + new Date().getTime();
    // Determine the target column based on card type.
    var targetColumnId = cardType === 'Tasks' ? 'Doing' : 'Qualified';
    var targetColumn = document.getElementById(targetColumnId);

    // Get the current card owner (assumed to be stored globally or use a placeholder).
    var cardOwner = window.currentCardOwner || "You";

    // Format the current date for follow-up history.
    var currentDate = new Date();
    var day = String(currentDate.getDate()).padStart(2, '0');
    var monthNames = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
    var year = currentDate.getFullYear();
    // Create the follow-up history string.
    var formattedFollowUpHistory = "[" + monthNames[currentDate.getMonth()] + " " + day + ", " + year + "] ['" + cardOwner + "' has created this card]";
    // Bold the date portion.
    var boldFormattedFollowUpHistory = formattedFollowUpHistory.replace(
      /\[(\w{3}\s\d{2},\s\d{4})\]/,
      '<span style="font-weight: 600;">[$1]</span>'
    );

    if (targetColumn) {
      // Build the full temporary card HTML with all sections.
      var tempCard = document.createElement('div');
      tempCard.id = 'card-' + tempUniqueID;
      
      // If the card is special, mark it immediately.
      if (specialCard) {
        tempCard.setAttribute('data-special-card', "Special");
        tempCard.className = 'card special-card';
      } else {
        tempCard.setAttribute('data-special-card', "");
        tempCard.className = 'card';
      }
      tempCard.setAttribute('data-project', projectName);
      tempCard.setAttribute('data-card-owner', cardOwner);
      tempCard.setAttribute('data-created-date', currentDate.toLocaleDateString('en-US'));
      tempCard.setAttribute('data-topic', topic);
      tempCard.setAttribute('data-unique-id', tempUniqueID);

      tempCard.innerHTML = `
        <div class="card-tag">
          <input type="text" class="tag-input" value="" placeholder="Add tag" />
        </div>
        <div class="oem-progress-box">
          <!-- OEM progress placeholder (if applicable) -->
        </div>
        <div class="amount-box">
          <!-- Amount details placeholder -->
        </div>

          <div class="low-uppercard">
            <div class="project-name">${projectName}</div>
            <div class="topic-name">${topic}</div>
          </div>
        </div>
        <div class="text-card">
          <textarea placeholder="Add Record" style="height: 23px;"></textarea>
        </div>

          ${boldFormattedFollowUpHistory}
        </div>
        <div class="lower-card">
          <div class="low-lowercard">
            <div class="icon-group-left"></div>
            <div class="icon-group-right"></div>
          </div>
        </div>
      `;

      if (specialCard) {
        // For special cards, insert after the last special card if any exist.
        var specialCards = targetColumn.querySelectorAll('.card.special-card');
        if (specialCards.length > 0) {
          var lastSpecial = specialCards[specialCards.length - 1];
          lastSpecial.parentNode.insertBefore(tempCard, lastSpecial.nextSibling);
        } else {
          // If no special card exists, insert after the "amount" element.
          var amountEl = targetColumn.querySelector('.amount');
          if (amountEl && amountEl.nextSibling) {
            targetColumn.insertBefore(tempCard, amountEl.nextSibling);
          } else {
            // Fallback: insert at the beginning if "amount" is not found.
            targetColumn.insertBefore(tempCard, targetColumn.firstChild);
          }
        }
      } else {
        // For normal cards, simply append to the end.
        targetColumn.appendChild(tempCard);
      }
      
      // Immediately update the card count or any other UI elements.
      updateCardCount();
    }
    
    // Call the server function to add the new card to the sheet in the background.
    google.script.run.withSuccessHandler(function() {
      // Refresh the board once the server update completes.
      fetchOpportunitiesData();
    }).addNewCardtoOpportunity(projectName, cardType, topic, specialCard);

    // Clear the form fields after submission.
    projectNameDropdown.selectedIndex = 0;
    cardTypeDropdown.selectedIndex = 0;
    topicInput.value = '';
    specialCardCheckbox.checked = false;

    // Optionally, close the Add Card sidebar.
    var addButtonSidebar = document.getElementById("addButtonSidebar");
    addButtonSidebar.style.width = "0px";  
  } else {
    alert('Please fill out all fields before submitting.');
  }
}

function duedateSubmit() {
  collapseAllExpandedCards(); // Collapse all expanded cards
  var newDueDateInput = document.getElementById('newDueDate');
  if (!newDueDateInput) {
    console.error('newDueDate element not found');
    return;
  }

  var newDueDate = newDueDateInput.value;
  var dueDateReason = document.getElementById('DueDateReason').value.trim();
  var uniqueId = document.getElementById('info-sidebar').getAttribute('data-unique-id');

  if (!newDueDate || dueDateReason === '') {
    alert('Please select a date and enter a reason for the due date change.');
    return;
  }

  var formattedDate = newDueDate.split('-').reverse().join('/');
  var card = document.querySelector(`.card[data-unique-id="${uniqueId}"]`);
  var existingDueDate = card.getAttribute('data-due-date');
  
  // Determine the action type
  var actionType = existingDueDate ? 'changed due date' : 'added due date';

  google.script.run.withSuccessHandler(function(displayName) {
    // Format the new history entry
    const now = new Date();
    const formattedDateForHistory = now.toLocaleDateString('en-US', { day: '2-digit', month: 'short', year: 'numeric' });
    var newHistoryEntry = `[${formattedDateForHistory}] ['${displayName}' ${actionType} to ${formattedDate}] ${dueDateReason}`;

    // Update the card in the front-end
    if (card) {
      card.setAttribute('data-due-date', formattedDate);

      // Update the due date icon in the card without removing other icons
      var dueDateIconHtml = getDueDateIcon(formattedDate, false); // Generate the due date icon HTML
      var iconGroupRight = card.querySelector('.icon-group-right');
      if (iconGroupRight) {
        // Find and replace only the due date container, keeping other icons intact
        var existingDueDateContainer = iconGroupRight.querySelector('.due-date-container');
        if (existingDueDateContainer) {
          existingDueDateContainer.outerHTML = dueDateIconHtml; // Replace the old due date container with the new one
        } else {
          // If there's no existing due date container, append the new one
          iconGroupRight.innerHTML += dueDateIconHtml;
        }
      }

      // Prepend the new history entry with a line break above it
      var followUpHistory = card.querySelector('.middle-card');
      followUpHistory.innerHTML = formatFollowUpHistory(newHistoryEntry + '\n\n' + followUpHistory.innerHTML.trim());

      // Update the card's data attributes with the new follow-up history
      card.setAttribute('data-follow-up-history', followUpHistory.innerHTML.trim());

      // Call updateCardColors to ensure the card color is updated based on the new due date
      updateCardColors();
    }

    // Update the server-side data in Google Sheets
    google.script.run.updateDueDateAndReason(uniqueId, formattedDate, dueDateReason, displayName);
  }).authorizedUser();

  displayTemporaryMessage("Due Date Modified Successfully.");
  closeInfoSidebar();
}

function setupInputChangeAmountField() {
  var inputChangeAmount = document.getElementById('inputChangeAmount');
  if (!inputChangeAmount) return;
  
  inputChangeAmount.addEventListener('input', function(event) {
    var cursorPosition = event.target.selectionStart;
    var originalLength = event.target.value.length;
    
    // Allow only digits and dot.
    let cleaned = event.target.value.replace(/[^0-9.]/g, '');
    
    // Remove extra dots (keep only the first one)
    const firstDotIndex = cleaned.indexOf('.');
    if (firstDotIndex !== -1) {
      cleaned = cleaned.substring(0, firstDotIndex + 1) +
                cleaned.substring(firstDotIndex + 1).replace(/\./g, '');
    }
    
    // Add comma formatting to the integer part while preserving decimals
    let parts = cleaned.split('.');
    parts[0] = parts[0].replace(/\B(?=(\d{3})+(?!\d))/g, ',');
    event.target.value = parts.join('.');
    
    var newLength = event.target.value.length;
    cursorPosition = cursorPosition + (newLength - originalLength);
    event.target.setSelectionRange(cursorPosition, cursorPosition);
  });
}

function editGoodsDate(uniqueId) {
  // Function to format date from "yyyy-mm-dd" to "dd/mm/yyyy"
  function formatDate(dateString) {
    if (dateString) {
      var parts = dateString.split('-');
      return parts[2] + '/' + parts[1] + '/' + parts[0];
    }
    return '';
  }

  var goodsReadyDate = formatDate(document.getElementById('inputGoodsReadyDate').value);
  var goodsLoadingDate = formatDate(document.getElementById('inputGoodsLoadingDate').value);
  var cardElement = document.querySelector(`.card[data-unique-id="${uniqueId}"]`);

  if (cardElement) {
    // Update the card element attributes
    cardElement.setAttribute('data-goods-ready-date', goodsReadyDate);
    cardElement.setAttribute('data-goods-loading-date', goodsLoadingDate);

    // Update the sidebar display for goods dates
    document.getElementById('inputGoodsReadyDate').value = goodsReadyDate.split('/').reverse().join('-');
    document.getElementById('inputGoodsLoadingDate').value = goodsLoadingDate.split('/').reverse().join('-');

    // Update the card text for the goods loading date icon
    var isEndColumn = cardElement.getAttribute('data-status') === 'End';
    var goodsLoadingDateIconHtml = getGoodsLoadingDateIcon(goodsLoadingDate, isEndColumn);

    // Find the container in the card to update the goods loading date icon
    var goodsLoadingDateContainer = cardElement.querySelector(".loading-date-container");
    if (goodsLoadingDateContainer) {
      // Replace the container's inner HTML with the updated icon HTML
      goodsLoadingDateContainer.innerHTML = goodsLoadingDateIconHtml;
    } else if (goodsLoadingDateIconHtml) {
      // If the container doesn't exist and an icon needs to be displayed, insert it into the card
      var iconGroupRight = cardElement.querySelector(".icon-group-right");
      if (iconGroupRight) {
        iconGroupRight.insertAdjacentHTML('beforeend', goodsLoadingDateIconHtml);
      }
    }

    // Update the "progress-text" calculation
    const oemStartedDateStr = cardElement.getAttribute('data-fda-started-date');
    const endDateStr = goodsReadyDate || cardElement.getAttribute('data-goods-ready-date');

    const oemStartedDate = oemStartedDateStr ? new Date(oemStartedDateStr.split('/').reverse().join('-')) : null;
    const endDate = endDateStr ? new Date(endDateStr.split('/').reverse().join('-')) : null;

    let daysPassed = 0;

    var progressTextElement = cardElement.querySelector('.progress-text');

    if (oemStartedDate) {
      if (endDate) {
        // Calculate days between FDA-Started-Date and Goods-Ready-Date
        daysPassed = Math.floor((endDate - oemStartedDate) / (1000 * 60 * 60 * 24));
      } else {
        // Calculate days between FDA-Started-Date and today
        var today = new Date();
        daysPassed = Math.floor((today - oemStartedDate) / (1000 * 60 * 60 * 24));
      }

      var formattedDaysPassed = daysPassed.toString().padStart(2, '0');

      if (progressTextElement) {
        progressTextElement.textContent = `${formattedDaysPassed}/60`; // Update the progress text
        progressTextElement.style.display = 'block'; // Ensure it is visible
      } else {
        // If the progress-text element is missing, create it
        var progressBox = cardElement.querySelector('.progress-box');
        if (progressBox) {
          progressBox.insertAdjacentHTML(
            'beforeend',
            `<div class="progress-text">${formattedDaysPassed}/60</div>`
          );
        }
      }
    } else {
      // Hide the progress-text if no FDA-Started-Date is selected
      if (progressTextElement) {
        progressTextElement.style.display = 'none';
      }
    }

    // Update the backend with the new goods dates values
    google.script.run.updateGoodsDate(uniqueId, goodsReadyDate, goodsLoadingDate);
    displayTemporaryMessage("Date Modified Successfully.");
  }
}

// Function to format the amount in "K" and "M"
function formatAmount(amount) {
  if (amount >= 1000000) {
    return (amount / 1000000).toFixed(2) + 'M';
  } else if (amount >= 1000) {
    return Math.round(amount / 1000) + 'K';
  } else {
    return amount.toString();
  }
}

function disableButtons(container, disable, activeButton = null, activeButtonClass = '') {
  const buttons = container.querySelectorAll('button');

    button.disabled = disable;

      button.classList.add('disabled');
      if (button === activeButton && activeButtonClass) {
        button.classList.add(activeButtonClass);
      }
    } else {
      button.classList.remove('disabled');
      button.classList.remove(activeButtonClass); 
    }
  });
}

document.addEventListener('DOMContentLoaded', function() {
    const hashtagIcon = document.querySelector('.hashtag-icon');
    const hashtagDropdown = document.querySelector('.hashtag-dropdown');
    const searchInput = document.querySelector('.searchbar');
    const cardArea = document.querySelector('.status-columns'); // Adjust selector as needed to cover card area

    // Toggle dropdown for hashtags
    hashtagIcon.addEventListener('click', function(event) {
        hashtagDropdown.style.display = hashtagDropdown.style.display === 'none' ? 'block' : 'none';
        populateHashtags(); // Function to populate the hashtags if not already populated
        event.stopPropagation(); // Prevent document click from affecting this
    });

    // Click outside the dropdowns to close
    document.addEventListener('click', function(event) {
        if (!hashtagDropdown.contains(event.target) && !hashtagIcon.contains(event.target)) {
            hashtagDropdown.style.display = 'none';
        }

        // Clear search bar only if clicking outside of the search and card areas
        if (!searchInput.contains(event.target) && !cardArea.contains(event.target) &&
            !hashtagDropdown.contains(event.target) && !atDropdown.contains(event.target)) {
            searchInput.value = '';
            handleSearch(); // Function to handle search functionality
        }
    });
});



function populateHashtags() {
    const text = document.body.innerText; // Get all text in the body
    const regex = /#(\w+)/g; // Regex to find hashtags
    let match;
    const dropdown = document.querySelector('.hashtag-dropdown');
    dropdown.innerHTML = ''; // Clear previous items

    const uniqueHashtags = new Set();
    while ((match = regex.exec(text)) !== null) {
        uniqueHashtags.add(match[0]);
    }

    uniqueHashtags.forEach(hashtag => {
        const div = document.createElement('div');
        div.className = 'hashtag';
        div.textContent = hashtag;
        div.onclick = () => {
            filterHashtag(hashtag);
            dropdown.style.display = 'none'; // Optionally close the dropdown on selection
        };
        dropdown.appendChild(div);
    });
}

function filterHashtag(hashtag) {
  var searchInput = document.querySelector('.searchbarpipeline');
  searchInput.value = hashtag; // Set the hashtag into the search bar
  searchInput.dispatchEvent(new Event('input', { bubbles: true })); // Trigger the input event
  updateCardCount(); // Update the card count after filtering
}

function populateAtDropdown(owners) {
    const dropdown = document.querySelector('.card-owner-dropdown');
    dropdown.innerHTML = '<option value="">ALL</option>'; // Reset and add default option
    
    const uniqueOwners = new Set(); // Create a set to hold unique owner names
    owners.forEach(owner => {
        if (owner && !uniqueOwners.has(owner)) { // Check if the owner is not already added
            uniqueOwners.add(owner); // Add owner to the set
            const option = document.createElement('option');
            option.value = owner;
            option.textContent = owner;
            dropdown.appendChild(option);
        }
    });
}

function filterAt(ownerName) {
  ownerName = ownerName.toLowerCase();
  var cards = document.querySelectorAll('.card');
  cards.forEach(card => {
      var cardOwner = card.getAttribute('data-card-owner').toLowerCase();
      if (ownerName === '' || cardOwner === ownerName) {
          card.style.display = '';
      } else {
          card.style.display = 'none';
      }
  });
  updateCardCount(); // Recalculate the count after filtering
}

// Function to format numbers with thousand separators and remove decimals if zero
function formatNumber(number) {
    if (number) {
        var formattedNumber = number.toLocaleString('en-US', {
            minimumFractionDigits: number % 1 === 0 ? 0 : 2,
            maximumFractionDigits: 2
        });
        return formattedNumber;
    }
    return '';
}

function displayOemProgressSection(uniqueId, oemData, container) {
  if (!oemData || !Array.isArray(oemData)) return;
  
  // The three progress types we want to show
  const progressTypes = ["RD Section", "Label / Packaging Section", "Production Section"];
  
  // Filter data to only those Types
  const progressData = oemData.filter(item => progressTypes.includes(item["Type"]));
  
  // Group by Type
  const groupedProgress = {};
  progressData.forEach(item => {
    const type = item["Type"];
    if (!groupedProgress[type]) groupedProgress[type] = [];
    groupedProgress[type].push(item);
  });
  
  // Clear the container and set some basic styling
  container.innerHTML = '';
  container.style.margin = "10px";
  
  // For each relevant Type, show a sub-section with checkboxes
  progressTypes.forEach(type => {
    if (groupedProgress[type] && groupedProgress[type].length > 0) {
      const sectionDiv = document.createElement('div');
      sectionDiv.className = 'oem-progress-group';
      
      const header = document.createElement('h3');
      header.textContent = type;
      sectionDiv.appendChild(header);
      
      groupedProgress[type].forEach(item => {
        const taskDiv = document.createElement('div');
        taskDiv.className = 'oem-task-item';
        taskDiv.style.display = 'flex';
        taskDiv.style.alignItems = 'center';
        taskDiv.style.marginBottom = '5px';
        
        // Create a checkbox
        const checkbox = document.createElement('input');
        checkbox.type = 'checkbox';
        checkbox.style.marginRight = '10px';
        checkbox.setAttribute("data-code", item["Code"]);
        
        // If the cell for this uniqueId is "Yes", check it
        const currentValue = (item[uniqueId] || "").trim();
        if (currentValue.toLowerCase() === "yes") {
          checkbox.checked = true;
        }
        
        // When changed, call server function
        checkbox.addEventListener('change', function() {
          const newValue = this.checked ? "Yes" : "";
          google.script.run.withSuccessHandler(function() {
            displayTemporaryMessage("OEM Task updated.");
          }).updateOemProgressTask(uniqueId, item["Code"], newValue);
        });
        
        const taskSpan = document.createElement('span');
        taskSpan.textContent = item["Field Name (EN)"] || "";
        
        taskDiv.appendChild(checkbox);
        taskDiv.appendChild(taskSpan);
        sectionDiv.appendChild(taskDiv);
      });
      
      container.appendChild(sectionDiv);
    }
  });
}

//---------------------------------- Utility Functions ----------------------------------
function saveOemTextareas() {
  const uniqueId = window.currentOemUniqueId;
  if (!uniqueId) {
    alert("Unique ID not set.");
    return;
  }

  const updates = [];

  // --- Group 1: Standard OEM Textareas (non-component ones) ---
  const standardTextareas = document.querySelectorAll('.oem-textarea:not(.oem-component-textarea)');
  standardTextareas.forEach(textarea => {
    const code = textarea.getAttribute('data-code');
    const value = textarea.value;
    updates.push({ uniqueId: uniqueId, code: code, value: value });
  });

  // --- Group 2: OEM Progress Checkboxes ---
  const progressCheckboxes = document.querySelectorAll('.oem-progress-group input[type="checkbox"]');
  progressCheckboxes.forEach(checkbox => {
    const code = checkbox.getAttribute('data-code');
    if (!code) return;
    const value = checkbox.checked ? "Yes" : "";
    updates.push({ uniqueId: uniqueId, code: code, value: value });
  });

  // --- Group 3: OEM Component Rows ---
  const uiGroup3Updates = {}; // mapping from code to combined value
  const uiGroup3Selects = document.querySelectorAll('.oem-component-section .oem-field-select');
  uiGroup3Selects.forEach(select => {
    const row = select.closest('tr');
    if (!row) return;
    const selectedOption = select.options[select.selectedIndex];
    const code = selectedOption ? selectedOption.getAttribute('data-code') : null;
    if (!code) return;
    const priceInput = row.querySelector('.oem-price-input');
    const price = priceInput ? priceInput.value.trim() : "";
    const textarea = row.querySelector('.oem-component-textarea');
    const detail = textarea ? textarea.value.trim() : "";
    let combinedValue = "";
    if (price && detail) {
      combinedValue = price + ", " + detail;
    } else if (price) {
      combinedValue = price;
    } else if (detail) {
      combinedValue = ", " + detail;
    }
    uiGroup3Updates[code] = combinedValue;
  });
  for (const code in uiGroup3Updates) {
    updates.push({ uniqueId: uniqueId, code: code, value: uiGroup3Updates[code] });
  }
  // Clear any group 3 items not present in the UI.
  const group3Types = ["เนื้อผลิตภัณฑ์", "บรรจุภัณฑ์", "ลัง", "ค่าแรง"];
  const allGroup3Codes = new Set();
  if (window.oemDataGlobal && window.oemDataGlobal.length > 0) {
    window.oemDataGlobal.forEach(item => {
      if (group3Types.includes(item["Type"])) {
        allGroup3Codes.add(item["Code"]);
      }
    });
  }
  allGroup3Codes.forEach(code => {
    if (!(code in uiGroup3Updates)) {
      updates.push({ uniqueId: uniqueId, code: code, value: "" });
    }
  });

  // ----- Additional: Update Reference Product and Volume -----
  if (window.oemDataGlobal && window.oemDataGlobal.length > 0) {
    const refProductRow = window.oemDataGlobal.find(item =>
      item["Type"] === "Referrence-Product" &&
      item["Field Name (EN)"].trim() === "Referred Product"
    );
    const refVolumeRow = window.oemDataGlobal.find(item =>
      item["Type"] === "Referrence-Product" &&
      item["Field Name (EN)"].trim() === "Referred Volume"
    );
    const refProductDropdown = document.querySelector('.reference-product-dropdown-select');
    const refProductValue = refProductDropdown ? refProductDropdown.value : "";
    const refVolumeInput = document.getElementById('ref-product-volume');
    const refVolumeValue = refVolumeInput ? refVolumeInput.value : "";
    if (refProductRow && refProductRow["Code"]) {
      updates.push({ uniqueId: uniqueId, code: refProductRow["Code"], value: refProductValue });
    }
    if (refVolumeRow && refVolumeRow["Code"]) {
      updates.push({ uniqueId: uniqueId, code: refVolumeRow["Code"], value: refVolumeValue });
    }
  }

  // ----- New: Update the approval field from oem-price-approve, if in edit mode -----
  const approveInput = document.querySelector('#oem-price-approve .oem-edit-input');
  if (approveInput) {
    const approveText = approveInput.value.trim();
    if (window.oemDataGlobal && window.oemDataGlobal.length > 0) {
      const approveRow = window.oemDataGlobal.find(item =>
        item["Type"] === "Approve" && item["Field Name (EN)"].trim() === "Approve"
      );
      if (approveRow && approveRow["Code"]) {
        const approveCode = approveRow["Code"];
        updates.push({ uniqueId: uniqueId, code: approveCode, value: approveText });
      }
    }
  }


  // Send all updates in one batch call.
  google.script.run.withSuccessHandler(function() {
    google.script.run.withSuccessHandler(function(oemData) {
      window.oemDataGlobal = oemData;
      // Refresh the OEM modal so that the price structure wrapper and all sections are re-rendered.
      showOemModal(uniqueId);
      displayTemporaryMessage("Data Saved");
    }).getOemSpec();
  }).batchUpdateOemSpecCells(updates);
}

function printOemPriceStructure() {
  // Get the Price Structure wrapper.
  const priceWrapper = document.getElementById('oemPriceStructureWrapper');
  if (!priceWrapper) {
    alert("No Price Structure available to print.");
    return;
  }
  
  // 1) Clone the Price Structure wrapper.
  const clonedPriceWrapper = priceWrapper.cloneNode(true);

  // 2) Sync all <select> current values into the clone
  const origSelects  = priceWrapper.querySelectorAll('select.oem-field-select, select.reference-product-dropdown-select');
  const cloneSelects = clonedPriceWrapper.querySelectorAll('select.oem-field-select, select.reference-product-dropdown-select');
  origSelects.forEach((orig, i) => {
    if (cloneSelects[i]) {
      cloneSelects[i].value = orig.value;
    }
  });

  // 3) Convert the original donut chart canvas to an image.
  const originalCanvas = document.getElementById('oemDonutChart');
  let imageLoaded = false;
  if (originalCanvas) {
    const imageUrl = originalCanvas.toDataURL("image/png");
    const clonedCanvas = clonedPriceWrapper.querySelector('#oemDonutChart');
    if (clonedCanvas) {
      const image = document.createElement('img');
      image.src = imageUrl;
      image.width = clonedCanvas.width;
      image.height = clonedCanvas.height;

      image.onload = function() {
        clonedCanvas.parentNode.replaceChild(image, clonedCanvas);
        imageLoaded = true;
        initiatePrint();
      };

      setTimeout(function() {
        if (!imageLoaded) {
          clonedCanvas.parentNode.replaceChild(image, clonedCanvas);
          initiatePrint();
        }
      }, 500);
    }
  } else {
    // No canvas found—print immediately.
    initiatePrint();
  }

  // 4) Clone the topic box into a print container.
  const navBar = document.getElementById('oemNavBar');
  const topicBox = navBar ? navBar.querySelector('.oem-topic-box') : null;
  let clonedTopicBox = null;
  if (topicBox) {
    clonedTopicBox = topicBox.cloneNode(true);
    clonedTopicBox.classList.add('cloned-topic-box');
    clonedTopicBox.style.marginBottom = '20px';
    clonedTopicBox.style.width = '100%';
  }

  // 5) Build the full print container.
  const printContainer = document.createElement('div');
  printContainer.id = 'printPriceStructureContainer';
  printContainer.style.width = '100%';
  if (clonedTopicBox) printContainer.appendChild(clonedTopicBox);
  clonedPriceWrapper.style.margin = '0 auto';
  printContainer.appendChild(clonedPriceWrapper);

  // 6) Append signature block.
  const signatureDiv = document.createElement('div');
  signatureDiv.style.width = '30%';
  signatureDiv.style.margin = '150px auto 0';
  signatureDiv.style.borderTop = '1px solid rgba(0,0,0,0.2)';
  signatureDiv.style.paddingTop = '10px';
  signatureDiv.style.textAlign = 'center';
  const now = new Date();
  const day = ("0" + now.getDate()).slice(-2);
  const month = now.toLocaleString('default', { month: 'short' });
  const year = now.getFullYear();
  signatureDiv.innerHTML = `
    <div style="font-weight: 300; font-size: 16px;">Thanisorn Rangsimawong</div>
    <div>${day} ${month} ${year}</div>
  `;
  printContainer.appendChild(signatureDiv);

  // 7) Append to body.
  document.body.appendChild(printContainer);

  // 8) Print helper.
  function initiatePrint() {
    setTimeout(() => {
      const originalStyles = [];
      Array.from(document.body.children).forEach(child => {
        if (child.id !== 'printPriceStructureContainer') {
          originalStyles.push({ el: child, disp: child.style.display });
          child.style.display = 'none';
        }
      });
      window.print();
      originalStyles.forEach(item => {
        item.el.style.display = item.disp;
      });
      document.body.removeChild(printContainer);
    }, 100);
  }
}

function getOemProgress(uniqueId) {
  // Allowed progress types (as used in your OEM-Progress-Section)
  const allowedTypes = ["RD Section", "Label / Packaging Section", "Production Section"];
  
  let totalCount = 0, checkedCount = 0;
  const container = document.getElementById('OEM-Progress-Section');
  
  if (container) {
    // Find all groups that represent OEM progress tasks
    const groups = container.querySelectorAll(".oem-progress-group");
    groups.forEach(group => {
      // Check the group header text to ensure it's one of the allowed types
      const header = group.querySelector("h3");
      if (header && allowedTypes.includes(header.textContent.trim())) {
        const checkboxes = group.querySelectorAll("input[type='checkbox']");
        checkboxes.forEach(cb => {
          totalCount++;
          if (cb.checked) {
            checkedCount++;
          }
        });
      }
    });
  }
  
  // If we found any valid checkboxes, compute percentage based on them
  if (totalCount > 0) {
    const percentage = Math.round((checkedCount / totalCount) * 100);
    return percentage + "%";
  }
  
  // Fallback: use the global OEM data but only for allowed types.
  if (!window.oemDataGlobal || !Array.isArray(window.oemDataGlobal)) return "";
  let total = 0, filled = 0;
  window.oemDataGlobal.forEach(item => {
    // Only consider rows whose Type is in allowedTypes
    if (allowedTypes.includes(item["Type"]) && item.hasOwnProperty(uniqueId)) {
      total++;
      if (item[uniqueId].trim() !== "") {
        filled++;
      }
    }
  });
  if (total > 0) {
    const percentage = Math.round((filled / total) * 100);
    return percentage + "%";
  }
  
  // If no data is available, return empty string to trigger the plus icon.
  return "";
}

  // Recalculate totals, row percentages, and type percentages.
  function recalcAndRenderPriceStructure() {
    let total = 0;
    const sumsByType = {};
    componentTypes.forEach(ct => { sumsByType[ct.key] = 0; });

    componentTypes.forEach(ct => {
      const section = rightSide.querySelector(`[data-type="${ct.key}"]`);
      if (!section) return;
      const rows = section.querySelectorAll('tbody tr');
      let typeSum = 0;
      rows.forEach(r => {
        const priceInput = r.querySelector('.oem-price-input');
        const val = parseFloat(priceInput.value) || 0;
        typeSum += val;
      });
      sumsByType[ct.key] = typeSum;
      total += typeSum;
    });

    componentTypes.forEach(ct => {
      const section = rightSide.querySelector(`[data-type="${ct.key}"]`);
      if (!section) return;
      const typeHeaderPct = section.querySelector('.oem-type-percentage');
      const rows = section.querySelectorAll('tbody tr');
      rows.forEach(r => {
        const priceInput = r.querySelector('.oem-price-input');
        const val = parseFloat(priceInput.value) || 0;
        const pctTd = r.querySelector('.oem-row-percent');
        let rowPct = (total > 0) ? (val / total) * 100 : 0;
        pctTd.textContent = rowPct.toFixed(1) + '%';
      });
      let typePct = (total > 0) ? (sumsByType[ct.key] / total) * 100 : 0;
      if (typeHeaderPct) {
        typeHeaderPct.textContent = typePct.toFixed(1) + '%';
      }
    });

    // Update total.
    const totalPriceEl = document.getElementById('oem-total-price');
    if (totalPriceEl) {
      totalPriceEl.textContent = total.toFixed(2);
    }

    // Now update the Chart.js donut.
    updateDonutChart(sumsByType, total);
  }

  // CHART.JS Donut Implementation.

      sumsByType['เนื้อผลิตภัณฑ์'] || 0,
      sumsByType['บรรจุภัณฑ์'] || 0,
      sumsByType['ลัง'] || 0,
      sumsByType['ค่าแรง'] || 0
    ];

      '#B689B2', // Formula
      '#D9A0B9', // Container
      '#F2B8B9', // Carton
      '#FFE2C8'  // Wage
    ];
    const labels = [
      'Formula',
      'Container',
      'Carton',
      'Wage'
    ];

    const canvas = document.getElementById('oemDonutChart');
    if (!canvas) return;

    if (donutChartInstance) {
      donutChartInstance.data.datasets[0].data = dataValues;
      donutChartInstance.update();
    } else {
      donutChartInstance = new Chart(canvas, {
        type: 'doughnut',
        data: {
          labels: labels,
          datasets: [{
            data: dataValues,
            backgroundColor: colors,
            borderColor: '#f9f9f9',
            borderWidth: 2
          }]
        },
        options: {
          responsive: true,
          maintainAspectRatio: false,
          cutout: '60%',
          plugins: {
            legend: { display: false },
            tooltip: {
              callbacks: {
                label: function(context) {
                  const label = context.label || '';
                  const val = context.parsed || 0;
                  return `${label}: ${val.toFixed(2)}`;
                }
              }
            }
          }
        }
      });
    }
  }
}

function updateComponentSelectOptions() {
  // If global OEM data is available, update the reference product and volume values once.
  if (window.oemDataGlobal && window.oemDataGlobal.length > 0) {
    const refProductRow = window.oemDataGlobal.find(item =>
      item["Type"] === "Referrence-Product" &&
      item["Field Name (EN)"].trim() === "Referred Product"
    );
    const refVolumeRow = window.oemDataGlobal.find(item =>
      item["Type"] === "Referrence-Product" &&
      item["Field Name (EN)"].trim() === "Referred Volume"
    );
    
    if (refProductRow && refProductRow[window.currentOemUniqueId] !== undefined) {
      const refProductValue = refProductRow[window.currentOemUniqueId];
      const dropdown = document.querySelector('.reference-product-dropdown-select');
      if (dropdown) {
        dropdown.value = refProductValue;
      }
    }
    
    if (refVolumeRow && refVolumeRow[window.currentOemUniqueId] !== undefined) {
      const refVolumeValue = refVolumeRow[window.currentOemUniqueId];
      const volumeInput = document.getElementById('ref-product-volume');
      if (volumeInput) {
        volumeInput.value = refVolumeValue;
        volumeInput.dispatchEvent(new Event('input'));
      }
    }
    
    // If a reference product is already selected, display the reference table.
    // Note: handleReferenceProductChange may override the volume value,
    // so reapply the OEM spec volume immediately after.
    const dropdown = document.querySelector('.reference-product-dropdown-select');
    if (dropdown && dropdown.value) {
      handleReferenceProductChange();
      if (refVolumeRow && refVolumeRow[window.currentOemUniqueId] !== undefined) {
        const refVolumeValue = refVolumeRow[window.currentOemUniqueId];
        const volumeInput = document.getElementById('ref-product-volume');
        if (volumeInput) {
          volumeInput.value = refVolumeValue;
          volumeInput.dispatchEvent(new Event('input'));
        }
      }
    }
  }
  
  // For each OEM component section (each section represents a type, e.g. "เนื้อผลิตภัณฑ์")
  const sections = document.querySelectorAll('.oem-component-section');
  sections.forEach(section => {
    const type = section.getAttribute('data-type');
    
    // Build a complete list of available options for this type from your global data.
    const allOptions = [];
    if (window.oemDataGlobal && window.oemDataGlobal.length > 0) {
      window.oemDataGlobal.forEach(item => {
        // Only add items matching the current section type (and excluding reference product rows)
        if (item["Type"] === type) {
          allOptions.push({ fieldName: item["Field Name (EN)"], code: item["Code"] });
        }
      });
    }
    
    // Get all select elements in this section
    const selects = section.querySelectorAll('.oem-field-select');
    // Gather current selections across the section (ignoring blank values)
    const selectedValues = Array.from(selects)
      .map(sel => sel.value)
      .filter(val => val !== '');
      
    // For each select, rebuild its options list.
    selects.forEach(sel => {
      const currentValue = sel.value; // preserve the current selection
      // Clear current options
      sel.innerHTML = "";
      // Add default option
      const defaultOpt = document.createElement('option');
      defaultOpt.value = '';
      defaultOpt.textContent = '--';
      sel.appendChild(defaultOpt);
      
      // Add each option from allOptions if:
      // - It is already selected in this select OR
      // - It is not selected in any other select in this section.
      allOptions.forEach(optData => {
        if (optData.fieldName === currentValue || !selectedValues.includes(optData.fieldName)) {
          const opt = document.createElement('option');
          opt.value = optData.fieldName;
          opt.setAttribute('data-code', optData.code);
          opt.textContent = optData.fieldName;
          sel.appendChild(opt);
        }
      });
      // Restore current selection (if it still exists)
      sel.value = currentValue;
    });
  });
}

function fetchOemDataForReference() {
  google.script.run.withSuccessHandler(function(data) {
    oemDataCache = data;
    populateReferenceProductDropdown(oemDataCache);
  }).getOemData();
}
document.addEventListener('DOMContentLoaded', fetchOemDataForReference);

function populateReferenceProductDropdown(oemData) {
  var dropdown = document.querySelector('.reference-product-dropdown-select');
  if (!dropdown) return;
  dropdown.innerHTML = '';
  var defaultOption = document.createElement('option');
  defaultOption.value = '';
  defaultOption.textContent = '-- Reference Product --';
  dropdown.appendChild(defaultOption);
  var referenceProducts = new Set();
  oemData.forEach(function(item) {
    if (item['Reference-Product'] && item['Reference-Product'].trim() !== '') {
      referenceProducts.add(item['Reference-Product'].trim());
    }
  });
  referenceProducts.forEach(function(product) {
    var option = document.createElement('option');
    option.value = product;
    option.textContent = product;
    dropdown.appendChild(option);
  });
}
  
function handleReferenceProductChange() {
  // Get the selected value from the dropdown
  var dropdown = document.querySelector('.reference-product-dropdown-select');
  var selected = dropdown.value;
  
  // Remove any existing reference section
  var existingSection = document.getElementById('oem-ref-table-section');
  if (existingSection) existingSection.remove();
  if (!selected) return; // Nothing to do if no option selected
  
  // Create a new section under the reference product container
  var container = document.querySelector('.oem-reference-product-container');
  var section = document.createElement('div');
  section.id = 'oem-ref-table-section';
  
  // Create the volume input container with label
  var volumeDiv = document.createElement('div');
  volumeDiv.className = 'ref-product-volume-container';
  volumeDiv.innerHTML =
    '<label for="ref-product-volume">อ้างอิงปริมาตร: </label>' +
    '<input type="number" id="ref-product-volume" value="" />';
  section.appendChild(volumeDiv);
  
  // Create the table container for the reference table
  var tableDiv = document.createElement('div');
  tableDiv.className = 'oem-ref-table';
  section.appendChild(tableDiv);
  
  // Append the section to the container
  container.appendChild(section);
  
  // Ensure the global OEM data cache exists
  if (!oemDataCache) return;
  
  // Find the matched row from OEM data based on "Reference-Product"
  let matchedRow = null;
  oemDataCache.forEach(function(row) {
    if (row['Reference-Product'] && row['Reference-Product'].trim() === selected) {
      matchedRow = row;
    }
  });
  if (!matchedRow) return;
  
  // Set the volume input's value from the "Volume" column
  const volumeValue = matchedRow['Volume'] || 0;
  const volumeInput = document.getElementById('ref-product-volume');
  volumeInput.value = volumeValue;
  
  // Build the reference table
  const table = document.createElement('table');
  table.style.width = '100%';
  
  // Define columns to exclude when building the table
  const excludeKeys = new Set(["Type", "Code", "Field Name (EN)", "Reference-Product", "Volume"]);
  
  // Iterate over each key in the matched row
  for (let key in matchedRow) {
    if (excludeKeys.has(key)) continue;
    const cellValue = matchedRow[key];
    if (!cellValue || String(cellValue).trim() === '') continue;
    
    // Split the cell value expecting a format like "0.5, อ้างอิงจาก..."
    const parts = cellValue.split(',');
    const numericPart = parts[0].trim();
    const detailPart = parts.slice(1).join(',').trim();
    
    // Create a new row
    const tr = document.createElement('tr');
    
    // Create the left cell
    const tdLeft = document.createElement('td');
    if (key.trim() === "อ้างอิงสูตร") {
      tdLeft.innerHTML = `
        <span class="oem-ref-header">${key} [${numericPart} x ${volumeValue}]</span><br>
        <span class="oem-ref-detail">${detailPart}</span>
      `;
      // Mark this row as special and store the multiplier & key name for dynamic calculation
      tr.setAttribute('data-special', 'true');
      tr.setAttribute('data-multiplier', numericPart);
      tr.setAttribute('data-key', key.trim());
    } else {
      // Normal case
      tdLeft.innerHTML = `
        <span class="oem-ref-header">${key}</span><br>
        <span class="oem-ref-detail">${detailPart}</span>
      `;
    }
    tr.appendChild(tdLeft);
    
    // Create the right cell (numeric value)
    const tdRight = document.createElement('td');
    tdRight.style.textAlign = 'right';
    tdRight.className = 'ref-right';
    
    if (key.trim() === "อ้างอิงสูตร") {
      const multiplier = parseFloat(numericPart) || 0;
      tdRight.textContent = (multiplier * parseFloat(volumeInput.value)).toFixed(2);
    } else {
      tdRight.textContent = numericPart;
    }
    tr.appendChild(tdRight);
    
    table.appendChild(tr);
  }
  
  tableDiv.appendChild(table);
  
  // Attach an event listener to update the special ("อ้างอิงสูตร") row calculation
  volumeInput.addEventListener('input', function() {
    const newVolume = parseFloat(this.value) || 0;
    const specialRows = table.querySelectorAll('tr[data-special="true"]');
    specialRows.forEach(function(row) {
      const multiplier = parseFloat(row.getAttribute('data-multiplier')) || 0;
      const keyName = row.getAttribute('data-key') || "อ้างอิงสูตร";
      const tdRight = row.querySelector('.ref-right');
      const headerSpan = row.querySelector('.oem-ref-header');
      
      // Update the right cell
      if (tdRight) {
        tdRight.textContent = (multiplier * newVolume).toFixed(2);
      }
      
      // Update the bracketed text in the header
      if (headerSpan) {
        headerSpan.textContent = `${keyName} [${multiplier} x ${newVolume}]`;
      }
    });
  });
}

  
document.addEventListener('change', function(e) {
  if (e.target && e.target.classList.contains('reference-product-dropdown-select')) {
    handleReferenceProductChange();
  }
});
